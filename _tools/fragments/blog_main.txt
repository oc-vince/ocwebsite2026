/* Blog index: progressive reveal, "load more" batching and a plain-text search
   over the card index. No dependency on GSAP — works on its own. */
(function(){
  var STEP  = 12;
  var grid  = document.getElementById('blGrid');
  var more  = document.getElementById('blMore');
  var wrap  = document.getElementById('blMoreWrap');
  var empty = document.getElementById('blEmpty');
  var count = document.getElementById('blCount');
  var input = document.getElementById('blSearch');
  if (!grid) return;

  var cards   = Array.prototype.slice.call(grid.querySelectorAll('.bl-card'));
  var matched = cards.slice();
  var shown   = 0;

  var reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var io = ('IntersectionObserver' in window) && !reduced
    ? new IntersectionObserver(function(entries, obs){
        entries.forEach(function(e){
          if (!e.isIntersecting) return;
          e.target.classList.add('is-shown');
          obs.unobserve(e.target);
        });
      }, { rootMargin: '0px 0px -8% 0px' })
    : null;

  function reveal(el, i){
    if (!io) { el.classList.add('is-shown'); return; }
    el.style.transitionDelay = Math.min(i, 5) * 60 + 'ms';
    io.observe(el);
  }

  function render(reset){
    if (reset) { shown = 0; cards.forEach(function(c){ c.hidden = true; c.classList.remove('is-shown'); }); }
    var next = matched.slice(shown, shown + STEP);
    next.forEach(function(c, i){ c.hidden = false; reveal(c, i); });
    shown += next.length;
    wrap.hidden = shown >= matched.length;
    empty.classList.toggle('is-on', matched.length === 0);
    count.textContent = matched.length;
  }

  more.addEventListener('click', function(){
    var first = matched[shown];
    render(false);
    if (first) first.focus && first.focus();
  });

  var t = null;
  input.addEventListener('input', function(){
    clearTimeout(t);
    t = setTimeout(function(){
      var q = input.value.trim().toLowerCase();
      matched = q ? cards.filter(function(c){ return c.getAttribute('data-t').indexOf(q) > -1; }) : cards.slice();
      render(true);
    }, 160);
  });

  render(true);
})();
</script>
